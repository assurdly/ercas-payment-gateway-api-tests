function generateNigerianPhoneNumber() {
    const prefixes = [
        "0701", "0703", "0704", "0705", "0706", "0708",
        "0802", "0803", "0805", "0806", "0807", "0808", "0809",
        "0810", "0811", "0812", "0813", "0814", "0815", "0816",
        "0817", "0818",
        "0901", "0902", "0903", "0904", "0905", "0906", "0907",
        "0908", "0909",
        "0911", "0912", "0913", "0915", "0916"
    ];

    const prefix = prefixes[Math.floor(Math.random() * prefixes.length)];
    const remainingDigits = Math.floor(1000000 + Math.random() * 9000000);

    return prefix + remainingDigits;
}
