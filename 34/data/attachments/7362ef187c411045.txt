let jsonData = {};
pm.test("Response is valid JSON", function () {
    jsonData = pm.response.json();
    pm.expect(jsonData).to.be.an("object");
});

pm.test("Status code is 200", function () {
    pm.expect([200]).to.include(pm.response.code);
});

pm.test("Content-Type is application/json", function () {
    pm.expect(pm.response.headers.get("Content-Type") || "").to.include("application/json");
});

pm.test("Response time is acceptable", function () {
    pm.expect(pm.response.responseTime).to.be.below(2000);
});

pm.test("ErcasPay response envelope is present", function () {
    pm.expect(jsonData).to.have.property("requestSuccessful");
    pm.expect(jsonData).to.have.property("responseCode");
    pm.expect(jsonData).to.have.property("responseBody");
});

pm.test("Checkout URL is returned", function () {
    pm.expect(jsonData.responseBody?.checkoutUrl).to.be.a("string");
});
if (jsonData?.responseBody?.transactionReference) pm.environment.set("directTransactionRef", jsonData.responseBody.transactionReference);
if (jsonData?.responseBody?.transactionReference) pm.environment.set("transactionRef", jsonData.responseBody.transactionReference);
if (jsonData?.responseBody?.paymentReference) pm.environment.set("paymentReference", jsonData.responseBody.paymentReference);
if (jsonData?.responseBody?.checkoutUrl) pm.environment.set("checkoutUrl", jsonData.responseBody.checkoutUrl);
