let jsonData = {};
pm.test("Response is valid JSON", function () {
    jsonData = pm.response.json();
    pm.expect(jsonData).to.be.an("object");
});
// Inferred expected behaviour - confirm against API specification.
pm.test("Status code is 400", function () {
    pm.expect([400]).to.include(pm.response.code);
});
pm.test("Error response has failure details", function () {
    pm.expect(jsonData.errors.success).to.not.equal(true);
});
